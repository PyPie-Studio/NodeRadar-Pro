+                else if (RuntimeInformation.IsOSPlatform(OSPlatform.OSX))
+                {
+                    Process.Start(new ProcessStartInfo { FileName = "open", Arguments = $"\"{encodedUrl}\"", UseShellExecute = false });
+                }
+            }

+        }
